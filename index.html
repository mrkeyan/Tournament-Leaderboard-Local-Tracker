  <script type="text/babel">

    const { useState, useEffect, useMemo } = React;

    // Custom hook to sync state with the browser's localStorage

    function useLocalStorage(key, initialValue) {

      const [storedValue, setStoredValue] = useState(() => {

        try {

          const item = window.localStorage.getItem(key);

          return item ? JSON.parse(item) : initialValue;

        } catch (error) {

          console.warn(`Error reading localStorage key "${key}":`, error);

          return initialValue;

        }

      });

      useEffect(() => {

        try {

          window.localStorage.setItem(key, JSON.stringify(storedValue));

        } catch (error) {

          console.warn(`Error setting localStorage key "${key}":`, error);

        }

      }, [key, storedValue]);

      return [storedValue, setStoredValue];

    }

    function App() {

      // Replaced standard useState with useLocalStorage for persistent data

      const [activeTab, setActiveTab] = useLocalStorage('tourney_activeTab', 'dashboard');

      const [players, setPlayers] = useLocalStorage('tourney_players', []);

      const [bulkInput, setBulkInput] = useLocalStorage('tourney_bulkInput', '');

      const [numGames, setNumGames] = useLocalStorage('tourney_numGames', 3);

     

      const [tournaments, setTournaments] = useLocalStorage('tourney_tournaments', []);

      const [activeGameId, setActiveGameId] = useLocalStorage('tourney_activeGameId', 0);

      const shuffle = (array) => [...array].sort(() => Math.random() - 0.5);

      const generateRound1 = (pool) => {

        const activePool = pool.filter(p => !p.droppedOut);

        const shuffled = shuffle(activePool);

        const numMatches = shuffled.length <= 16 ? 4 : 5;

       

        const matches = Array.from({ length: numMatches }, (_, i) => ({

          id: `r1_m${i + 1}`,

          maxWinners: 2,

          participants: []

        }));

        shuffled.forEach((p, i) => {

          matches[i % numMatches].participants.push({ ...p, score: '', isWinner: false });

        });

        return matches;

      };

      const generateRound2 = (r1Winners) => {

        const matches = [];

        if (r1Winners.length === 8) {

          matches.push({ id: `r2_m1`, maxWinners: 2, participants: r1Winners.slice(0, 4).map(p => ({ ...p, score: '', isWinner: false })) });

          matches.push({ id: `r2_m2`, maxWinners: 2, participants: r1Winners.slice(4, 8).map(p => ({ ...p, score: '', isWinner: false })) });

        } else {

          matches.push({ id: `r2_m1`, maxWinners: 2, participants: r1Winners.slice(0, 4).map(p => ({ ...p, score: '', isWinner: false })) });

          matches.push({ id: `r2_m2`, maxWinners: 1, participants: r1Winners.slice(4, 7).map(p => ({ ...p, score: '', isWinner: false })) });

          matches.push({ id: `r2_m3`, maxWinners: 1, participants: r1Winners.slice(7, 10).map(p => ({ ...p, score: '', isWinner: false })) });

        }

        return matches;

      };

      const initializePool = () => {

        const names = bulkInput.split('\n').map(n => n.trim()).filter(n => n.length > 0);

        if (names.length < 16 || names.length > 20) {

          alert("Please enter between 16 and 20 players.");

          return;

        }

        const newPlayers = names.map((name, i) => ({ id: `p${i + 1}`, name, droppedOut: false }));

        setPlayers(newPlayers);

       

        const initialTournaments = Array.from({ length: numGames }, (_, i) => ({

          id: i,

          name: `Video Game ${i + 1}`,

          status: i === 0 ? "Live" : "Upcoming",

          rounds: [generateRound1(newPlayers)]

        }));

       

        setTournaments(initialTournaments);

        setActiveGameId(0);

      };

      const dropPlayer = (playerId) => {

        if(!confirm("Remove player permanently from all future matches? They will score 0 points.")) return;

        setPlayers(prev => prev.map(p => p.id === playerId ? { ...p, droppedOut: true } : p));

        setTournaments(prev => prev.map(t => {

          if (t.status === "Completed") return t;

          const newRounds = t.rounds.map(round => round.map(match => ({

            ...match,

            participants: match.participants.filter(p => p.id !== playerId)

          })));

          return { ...t, rounds: newRounds };

        }));

      };

      const updateTournamentName = (gameId, newName) => {

        setTournaments(prev => prev.map(t => t.id === gameId ? { ...t, name: newName } : t));

      };

      const reshuffleMatchups = (gameId) => {

        if (players.length === 0) return;

        setTournaments(prev => prev.map(t => {

          if (t.id === gameId) {

            return { ...t, rounds: [generateRound1(players)] };

          }

          return t;

        }));

      };

      const startTournament = (gameId) => {

        setTournaments(prev => prev.map(t => t.id === gameId ? { ...t, status: "Live" } : t));

      };

      const updateParticipant = (gameId, roundIndex, matchId, participantId, field, value) => {

        setTournaments(prev => prev.map(t => {

          if (t.id !== gameId) return t;

          const newRounds = [...t.rounds];

          const matchIndex = newRounds[roundIndex].findIndex(m => m.id === matchId);

          const match = { ...newRounds[roundIndex][matchIndex] };

         

          if (field === 'isWinner' && value === true) {

            const currentWinners = match.participants.filter(p => p.isWinner).length;

            if (currentWinners >= match.maxWinners) {

              alert(`Maximum of ${match.maxWinners} winner(s) allowed for this match.`);

              return t;

            }

          }

          match.participants = match.participants.map(p => p.id === participantId ? { ...p, [field]: value } : p);

          newRounds[roundIndex][matchIndex] = match;

          return { ...t, rounds: newRounds };

        }));

      };

      const advanceRound = (gameId) => {

        setTournaments(prev => prev.map(t => {

          if (t.id !== gameId) return t;

         

          const currentRoundIndex = t.rounds.length - 1;

          const currentRound = t.rounds[currentRoundIndex];

         

          const winners = [];

          let allMatchesResolved = true;

          currentRound.forEach(m => {

            const matchWinners = m.participants.filter(p => p.isWinner);

            if (matchWinners.length !== m.maxWinners) allMatchesResolved = false;

            winners.push(...matchWinners);

          });

          if (!allMatchesResolved) {

            alert("Please select the correct number of winners for all matches before advancing.");

            return t;

          }

          if (currentRoundIndex === 0) {

            return { ...t, rounds: [...t.rounds, generateRound2(winners)] };

          } else if (currentRoundIndex === 1) {

            const finalsMatch = {

              id: 'r3_finals',

              maxWinners: 1,

              participants: winners.map(p => ({ ...p, score: '', isWinner: false }))

            };

            return { ...t, rounds: [...t.rounds, [finalsMatch]] };

          } else if (currentRoundIndex === 2) {

            if (winners.length !== 1) {

              alert("Please select exactly 1 Grand Champion.");

              return t;

            }

            return { ...t, status: "Completed" };

          }

          return t;

        }));

      };

      const generateSuddenDeath = (tiedPlayerIds) => {

        const tiedPlayers = players.filter(p => tiedPlayerIds.includes(p.id));

        const sdGameId = tournaments.length;

       

        const sdMatch = {

          id: 'sd_finals',

          maxWinners: 1,

          participants: tiedPlayers.map(p => ({ ...p, score: '', isWinner: false }))

        };

        const sdTournament = {

          id: sdGameId,

          name: "Sudden Death Tiebreaker",

          status: "Live",

          rounds: [[sdMatch]]

        };

        setTournaments([...tournaments, sdTournament]);

        setActiveGameId(sdGameId);

        setActiveTab('brackets');

      };

      const leaderboard = useMemo(() => {

        if (players.length === 0) return [];

       

        let stats = players.reduce((acc, p) => {

          acc[p.id] = { id: p.id, name: p.name, gameScores: {}, total: 0, firsts: 0, finals: 0, droppedOut: p.droppedOut };

          return acc;

        }, {});

        tournaments.forEach(t => {

          if (t.status !== "Completed") return;

         

          const r1 = t.rounds[0].flatMap(m => m.participants);

          const r2 = t.rounds[1]?.flatMap(m => m.participants) || [];

          const r3 = t.rounds[2]?.[0]?.participants || [];

          players.forEach(p => {

            if (p.droppedOut) {

              stats[p.id].gameScores[t.id] = 0;

              return;

            }

            let pts = 0;

            const inR1 = r1.find(x => x.id === p.id);

            const inR2 = r2.find(x => x.id === p.id);

            const inR3 = r3.find(x => x.id === p.id);

            if (inR3) {

              stats[p.id].finals += 1;

              if (inR3.isWinner) { pts = 4; stats[p.id].firsts += 1; }

              else { pts = 3; }

            } else if (inR2) {

              pts = 2;

            } else if (inR1) {

              pts = 1;

            }

            stats[p.id].gameScores[t.id] = pts;

            stats[p.id].total += pts;

          });

        });

        return Object.values(stats).sort((a, b) => {

          if (a.droppedOut && !b.droppedOut) return 1;

          if (!a.droppedOut && b.droppedOut) return -1;

          if (b.total !== a.total) return b.total - a.total;

          if (b.firsts !== a.firsts) return b.firsts - a.firsts;

          return b.finals - a.finals;

        });

      }, [players, tournaments]);

      // Handle resetting local storage

      const resetEverything = () => {

        if(!confirm("Are you sure? This will wipe all saved tournament data from this browser.")) return;

        setPlayers([]);

        setTournaments([]);

        setBulkInput('');

        setNumGames(3);

        setActiveTab('dashboard');

        setActiveGameId(0);

        window.localStorage.clear();

      };

      return (

        <div className="min-h-screen flex flex-col md:flex-row font-sans text-slate-200">

         

          <div className="w-full md:w-64 bg-slate-800 border-r border-slate-700 flex flex-col">

            <div className="p-6 border-b border-slate-700">

              <h1 className="text-xl font-bold text-indigo-400">🏆 TourneyHub</h1>

            </div>

            <nav className="flex-1 p-4 space-y-2">

              <button onClick={() => setActiveTab('dashboard')} className={`w-full text-left px-4 py-3 rounded-lg ${activeTab === 'dashboard' ? 'bg-indigo-600' : 'hover:bg-slate-700'}`}>⚙️ Admin</button>

              <button onClick={() => setActiveTab('brackets')} className={`w-full text-left px-4 py-3 rounded-lg ${activeTab === 'brackets' ? 'bg-indigo-600' : 'hover:bg-slate-700'}`}>🌳 Brackets</button>

              <button onClick={() => setActiveTab('leaderboard')} className={`w-full text-left px-4 py-3 rounded-lg ${activeTab === 'leaderboard' ? 'bg-indigo-600' : 'hover:bg-slate-700'}`}>🏅 Leaderboard</button>

            </nav>

          </div>

          <div className="flex-1 p-6 md:p-8 overflow-auto">

           

            {activeTab === 'dashboard' && (

              <div className="max-w-3xl">

                <h2 className="text-2xl font-bold mb-6">Setup Tournament</h2>

                <div className="bg-slate-800 p-6 rounded-xl border border-slate-700">

                  {players.length === 0 ? (

                    <>

                      <div className="mb-4">

                        <label className="block text-sm text-slate-400 mb-1">Number of Games</label>

                        <input type="number" min="1" max="10" value={numGames} onChange={(e) => setNumGames(parseInt(e.target.value))} className="w-32 bg-slate-900 border border-slate-600 rounded p-2 text-white outline-none" />

                      </div>

                      <textarea

                        value={bulkInput}

                        onChange={(e) => setBulkInput(e.target.value)}

                        placeholder="Player 1\nPlayer 2..."

                        className="w-full h-48 bg-slate-900 border border-slate-600 rounded-lg p-4 outline-none mb-4"

                      />

                      <button onClick={initializePool} className="bg-indigo-600 hover:bg-indigo-500 px-6 py-2 rounded-lg font-medium">Create Tournaments</button>

                    </>

                  ) : (

                    <div className="space-y-4">

                      <p className="text-emerald-400 font-medium">✅ {players.length} Players Locked In</p>

                      <button onClick={resetEverything} className="text-red-400 text-sm hover:underline">Reset Everything (Danger)</button>

                    </div>

                  )}

                </div>

              </div>

            )}

            {activeTab === 'brackets' && players.length > 0 && (

              <div>

                <div className="flex gap-2 bg-slate-800 p-2 rounded-lg w-fit mb-6">

                  {tournaments.map(t => (

                    <button key={t.id} onClick={() => setActiveGameId(t.id)} className={`px-4 py-2 rounded-md ${activeGameId === t.id ? 'bg-indigo-600' : 'hover:bg-slate-700'}`}>

                      {t.name}

                    </button>

                  ))}

                </div>

                {tournaments.filter(t => t.id === activeGameId).map(t => (

                  <div key={t.id} className="bg-slate-800 border border-slate-700 rounded-xl p-6 overflow-x-auto">

                    <div className="flex justify-between items-center mb-6 min-w-[800px]">

                      <div>

                        <input

                          type="text"

                          value={t.name}

                          onChange={(e) => updateTournamentName(t.id, e.target.value)}

                          className="text-2xl font-bold text-indigo-300 bg-transparent border-b border-dashed border-indigo-700 focus:border-indigo-400 outline-none pb-1"

                        />

                        <div className="mt-2 text-sm text-slate-400">Status: {t.status}</div>

                      </div>

                     

                      <div className="flex gap-3">

                        {t.status === "Upcoming" && (

                          <>

                            <button onClick={() => reshuffleMatchups(t.id)} className="bg-slate-700 px-4 py-2 rounded-lg text-sm">🎲 Reshuffle</button>

                            <button onClick={() => startTournament(t.id)} className="bg-emerald-600 px-4 py-2 rounded-lg text-sm">▶️ Start</button>

                          </>

                        )}

                        {t.status === "Live" && (

                          <button onClick={() => advanceRound(t.id)} className="bg-indigo-600 px-4 py-2 rounded-lg text-sm">Advance Winners ➡️</button>

                        )}

                      </div>

                    </div>

                    <div className="flex gap-8 min-w-[800px]">

                      {t.rounds.map((round, rIndex) => (

                        <div key={rIndex} className="flex-1 min-w-[340px] space-y-4">

                          <h4 className="font-semibold text-slate-300 border-b border-slate-700 pb-2">

                            {rIndex === 0 ? "Round 1" : rIndex === 1 ? "Round 2" : "Grand Finals"}

                          </h4>

                         

                          {round.map((match, mIndex) => (

                            <div key={match.id} className="bg-slate-900 border border-slate-700 rounded-lg p-3">

                              <div className="text-xs text-slate-400 mb-2">Match {mIndex + 1} (Top {match.maxWinners})</div>

                              <div className="space-y-2">

                                {match.participants.map(p => {

                                  const isDropped = players.find(x => x.id === p.id)?.droppedOut;

                                  return (

                                    <div key={p.id} className={`flex items-center gap-2 p-2 rounded ${p.isWinner ? 'bg-indigo-900/40' : 'bg-slate-800'} ${isDropped ? 'opacity-50' : ''}`}>

                                      <input

                                        type="checkbox"

                                        checked={p.isWinner}

                                        disabled={t.status !== 'Live' || isDropped}

                                        onChange={(e) => updateParticipant(t.id, rIndex, match.id, p.id, 'isWinner', e.target.checked)}

                                        className="w-4 h-4"

                                      />

                                      <span className="flex-1 text-sm truncate">{p.name} {isDropped && '(Dropped)'}</span>

                                      <input

                                        type="text"

                                        placeholder="Score"

                                        value={p.score}

                                        disabled={t.status !== 'Live' || isDropped}

                                        onChange={(e) => updateParticipant(t.id, rIndex, match.id, p.id, 'score', e.target.value)}

                                        className="w-16 text-xs bg-slate-900 border border-slate-600 rounded px-1 outline-none"

                                      />

                                      <button onClick={() => dropPlayer(p.id)} className="text-red-400 text-xs hover:text-red-300">🗑️</button>

                                    </div>

                                  );

                                })}

                              </div>

                            </div>

                          ))}

                        </div>

                      ))}

                    </div>

                  </div>

                ))}

              </div>

            )}

            {activeTab === 'leaderboard' && (

              <div className="max-w-5xl">

                <div className="flex justify-between items-center mb-6">

                  <h2 className="text-2xl font-bold">Global Leaderboard</h2>

                  <button

                    onClick={() => {

                      const topScore = leaderboard[0]?.total;

                      const tiedIds = leaderboard.filter(p => p.total === topScore).map(p => p.id);

                      if (tiedIds.length > 1) generateSuddenDeath(tiedIds);

                      else alert("No tie for first place detected!");

                    }}

                    className="bg-amber-600 hover:bg-amber-500 text-white px-4 py-2 rounded-lg text-sm font-medium"

                    ⚔️ Sudden Death (Top Tie)
